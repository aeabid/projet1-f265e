::rule
@map المركّبات | الشّكل النّحويّ
- نعتيّ | منعوت + نعت (يتبعُه في كلّ شيء) | العُصْفورُ الصَّغيرُ · طاوِلَةٌ كَبيرَةٌ | star
- إضافيّ | مضاف + مضاف إليه (مجرور) | قَلَمُ الحِبْرِ · قَلَمُ مُحَمَّدٍ · قَلَمُهُ | tag
- بدليّ | الكلمتان = الشّيء نفسه | هٰذا الوَلَدُ · الخَليفَةُ عُمَرُ | mirror
- عطفيّ | وَ · فَـ · ثُمَّ · أَوْ | الوَلَدُ وَالبِنْتُ · شايًا أَوْ حَليبًا | chain
- بالجرّ | حرف جرّ + اسم مجرور | مِنَ البَيْتِ · إلى المَدْرَسَةِ · لَهُ | link
@parse
قَلَمُ=مضاف (بلا «ال» ولا تنوين) ; الحِبْرِ=مضاف إليه (مجرور)
التِّلْميذانِ=منعوت ; المُجْتَهِدانِ=نعت (يتبعُه)
@tree التّوابع | تتبعُ ما قبلها في الحركة
- النّعت | | جاءَ الطّالِبُ المُجْتَهِدُ | star
- البدل | | جاءَ هٰذا الطّالِبُ | mirror
- المعطوف | | جاءَ الطّالِبُ وَالطّالِبَةُ | chain
@remember المضافُ <b>لا تدخلُه «ال» ولا يُنوَّن</b> : قَلَمُ الحِبْرِ ✓ - القَلَمُ الحِبْرِ ✗ - قَلَمٌ الحِبْرِ ✗.

::ex classify 1 | أُصنّفُ المركّباتِ في الجدول
cols: نعتيّ | إضافيّ | بدليّ | عطفيّ | بالجرّ
نعتيّ: البَحْرُ الأَزْرَقُ - كِتابٌ مُفيدٌ
إضافيّ: بابُ القِسْمِ - حَقيبَةُ سَلْمى
بدليّ: تِلْكَ الشَّجَرَةُ - هٰذِهِ البِنْتُ
عطفيّ: القِطُّ وَالكَلْبُ - الخُبْزُ ثُمَّ الجُبْنُ
بالجرّ: في الحَديقَةِ - عَلى الطّاوِلَةِ

::ex match 1 | أربطُ كلَّ مركّبٍ بنوعه
قَميصٌ جَديدٌ => مركّبٌ نعتيّ
سَيّارَةُ أَبي => مركّبٌ إضافيّ
هٰؤُلاءِ الأَطْفالُ => مركّبٌ بدليّ
الشَّمْسُ وَالقَمَرُ => مركّبٌ عطفيّ
مِنَ السّوقِ => مركّبٌ بالجرّ

::ex table 2 | أُكوّنُ المركّبَ المطلوب
head: الكلمة | مركّبٌ نعتيّ | مركّبٌ إضافيّ | مركّبٌ بالجرّ
البَيْتُ | [البَيْتُ الكَبيرُ] | [بَيْتُ جَدّي] | [في البَيْتِ]
السَّيّارَةُ | [السَّيّارَةُ السَّريعَةُ] | [سَيّارَةُ أَبي] | [بِالسَّيّارَةِ]
الحَديقَةُ | [الحَديقَةُ الواسِعَةُ] | [حَديقَةُ المَدْرَسَةِ] | [إلى الحَديقَةِ]

::ex fix 2 | أُصلحُ الخطأ في المركّبات الإضافيّة
القَلَمُ التِّلْميذِ => قَلَمُ التِّلْميذِ
بابٌ المَنْزِلِ => بابُ المَنْزِلِ
الحَديقَةُ المَدْرَسَةِ => حَديقَةُ المَدْرَسَةِ
كِتابٌ سامي => كِتابُ سامي

::ex fill 2 | أُطابقُ النّعتَ مع المنعوت (أكتبُ الكلمةَ بين القوسين بالشّكل المناسب)
اِشْتَرَيْتُ ثَوْبًا [جَديدًا] ((جَديد))
هٰذِهِ زَهْرَةٌ [جَميلَةٌ] ((جَميل))
نَجَحَ التِّلْميذانِ [المُجْتَهِدانِ] ((مُجْتَهِد))
عادَتِ الطُّيورُ إلى أَعْشاشِها [الدّافِئَةِ] ((دافِئ))
شَكَرْنا المُمَرِّضاتِ [النَّشيطاتِ] ((نَشيط))

::ex mcq 2 | أختارُ الشّكلَ النّحويَّ للعبارة الملوّنة
جاءَ ==طالِبُ العِلْمِ==. {نعتيّ | *إضافيّ | بدليّ}
جاءَ ==الطّالِبُ المُجْتَهِدُ==. {*نعتيّ | إضافيّ | عطفيّ}
جاءَ ==الطّالِبُ وَالطّالِبَةُ==. {بدليّ | *عطفيّ | بالجرّ}
جاءَ ==هٰذا الطّالِبُ==. {نعتيّ | إضافيّ | *بدليّ}
مَرَرْتُ ==بِالطّالِبِ==. {إضافيّ | عطفيّ | *بالجرّ}

::ex order 2 | أرتّبُ الكلماتِ لأُكوّنَ جملًا، ثمّ أضعُ خطًّا تحت المركّب الإضافيّ
الحَديقَةِ / في / أَزْهارُ / تَفَتَّحَتْ
أَبي / سَيّارَةُ / بَيْضاءُ
المَدْرَسَةِ / مُديرُ / التَّلاميذَ / كافَأَ

::ex html 3 | أستخرجُ المركّباتِ من الفقرة وأُحدّدُ نوعَها
<div class="intro">في صَباحٍ مُشْمِسٍ، خَرَجَ الطِّفْلُ يوسُفُ مَعَ أَبيهِ إلى حَقْلِ الزَّيْتونِ. حَمَلَ يوسُفُ سَلَّةً صَغيرَةً، وَحَمَلَ الأَبُ السُّلَّمَ وَالعَصا. قالَ الأَبُ : «هٰذِهِ الشَّجَرَةُ قَديمَةٌ».</div>
